{{/* BlogPosting for article pages. Until now every URL on the site emitted only Person + WebSite, so no page ever declared a headline, a publish date, a modified date, an author or a section. Google had to infer all of it from the rendered HTML. Article structured data is also what makes a page eligible for article rich results and for the "updated" freshness signal on a series that gets revised. Built with dict + jsonify rather than hand-written JSON so that quotes and non-ASCII in titles and descriptions are escaped instead of producing invalid JSON-LD. */}} {{- if and .IsPage .Date (not .Params.noindex) }} {{- $desc := .Description | default .Summary | plainify | chomp }} {{- $img := "" }} {{- with .Params.thumbnail }} {{- with $.Resources.GetMatch . }} {{- $img = .Permalink }} {{- else }} {{- $img = absURL . }} {{- end }} {{- end }} {{- $schema := dict "@context" "https://schema.org" "@type" "BlogPosting" "headline" (.Title | plainify) "description" $desc "url" .Permalink "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "inLanguage" .Language.Locale "wordCount" .WordCount "author" (dict "@type" "Person" "name" "Andrius Mikonis" "url" "https://itohi.com") "publisher" (dict "@type" "Organization" "name" "ITOHI" "url" "https://itohi.com") }} {{- with $img }}{{ $schema = merge $schema (dict "image" (slice .)) }}{{ end }} {{- with .Params.keywords }}{{ $schema = merge $schema (dict "keywords" (delimit . ", ")) }}{{ end }} {{- with .Params.categories }}{{ $schema = merge $schema (dict "articleSection" (index . 0)) }}{{ end }} {{- with .Params.series }}{{ $schema = merge $schema (dict "isPartOf" (dict "@type" "CreativeWorkSeries" "name" (index . 0))) }}{{ end }} {{- end }}
